Control Flow

# If/Else/Elif
yedi age >= 18:
    print("Adult")
athawa age >= 13:
    print("Teen")
natra:
    print("Child")

# For Loop
ko_lagi item ma items:
    print(item)

ko_lagi i ma range(10):
    print(i)

# While Loop
jabasamma count < 10:
    count += 1

# Loop Control
rok      # break
agadi    # continue

Functions

# Basic Function
kaam add(a, b):
    firta a + b

# Default Parameters
kaam greet(name="World"):
    print(f"Namaste {name}!")

# Keyword Arguments
kaam create_user(name, age=21):
    print(f"User: {name}, Age: {age}")

# Recursion
kaam factorial(n):
    yedi n <= 1:
        firta 1
    firta n * factorial(n - 1)

Classes

# Basic Class
kakshya Person:
    kaam __init__(self, name, age):
        self.name = name
        self.age = age
    
    kaam greet(self):
        firta f"Namaste, I am {self.name}"

# Inheritance
kakshya Student extends Person:
    kaam __init__(self, name, age, grade):
        super().__init__(name, age)
        self.grade = grade

Error Handling

# Try/Except/Finally
koshish:
    result = risky_operation()
samau error:
    print(f"Error: {error}")
antya:
    print("Cleanup")

# Raise Exception
uthau ValueError("Invalid input")
